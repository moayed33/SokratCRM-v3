GSAP TARGET: #chartPieSlices 
// Targeting: function animateBar(barId) targeting #chartPieSlices
function animateBar(barId) {
  const targetSelector = `#chartPieSlices #bar-${barId}`;
  
  gsap.fromTo(targetSelector, {
    scaleY: 0,
    transformOrigin: "bottom center",
    opacity: 0.3
  }, {
    scaleY: 1,
    opacity: 1,
    duration: 0.65,
    ease: "back.out(1.7)",
    stagger: 0.08
  });
}
Target: #chartLinePath
SVG PATH SYNCHRONIZER 
// Target: #chartLinePath
function animateLinePath(pathSelector = "#chartLinePath") {
  const path = document.querySelector(pathSelector);
  const length = path.getTotalLength();
  
  gsap.set(path, { 
    strokeDasharray: length, 
    strokeDashoffset: length 
  });
  
  gsap.to(path, {
    strokeDashoffset: 0,
    duration: 1.2,
    ease: "power2.out",
    delay: 0.25
  });
}
